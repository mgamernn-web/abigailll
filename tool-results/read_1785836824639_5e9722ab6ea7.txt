   103→async function startPopcornShootTimer(game) {
   104→  if (game.mode !== GAME_MODE.POPCORN || game.state === GAME_STATE.ENDED) return;
   105→  if (game.shootTimer) { clearTimeout(game.shootTimer); game.shootTimer = null; }
   106→
   107→  const channel = game.channel;
   108→  if (!channel) return;
   109→
   110→  const gunHolder = game.players.get(game.gunHolder);
   111→  if (!gunHolder) return;
   112→
   113→  // Warning at 10 seconds before expiry
   114→  const warningTime = Math.max(0, (game.shootTimerLength - 10) * 1000);
   115→  let warningSent = false;
   116→
   117→  if (warningTime > 0) {
   118→    game._shootWarningTimer = setTimeout(async () => {
   119→      if (game.state === GAME_STATE.ENDED || game.gunHolder !== gunHolder.user.id) return;
   120→      warningSent = true;
   121→      try {
   122→        await channel.send({ embeds: [new EmbedBuilder()
   123→          .setColor(0xF39C12)
   124→          .setTitle('⏰ Shoot Timer Warning!')
   125→          .setDescription(`**${gunHolder.user.username}** has **10 seconds** left to shoot!\nUse \`w.shoot <number>\` now or you'll be eliminated!`)
   126→          .setTimestamp()] });
   127→      } catch (e) {}
   128→    }, warningTime);
   129→  }
   130→
   131→  // Main timer — gun holder is eliminated if they don't shoot in time
   132→  game.shootTimer = setTimeout(async () => {
   133→    if (game._shootWarningTimer) { clearTimeout(game._shootWarningTimer); game._shootWarningTimer = null; }
   134→    if (game.state === GAME_STATE.ENDED) return;
   135→    if (game.gunHolder !== gunHolder.user.id) return;  // gun already passed
   136→
   137→    // Gun holder didn't shoot — they are eliminated!
   138→    gunHolder.alive = false;
   139→
   140→    const timeoutEmbed = new EmbedBuilder()
   141→      .setColor(0xE74C3C)
   142→      .setTitle('⏰ Time\'s Up!')
   143→      .setDescription(
   144→        `**${gunHolder.user.username}** ran out of time and is eliminated!\n` +
   145→        `💀 **${gunHolder.user.username}** (${ROLE_EMOJI[gunHolder.role]} ${gunHolder.role}) couldn't pull the trigger!`
   146→      )
   147→      .setTimestamp();
   148→    await channel.send({ embeds: [timeoutEmbed] });
   149→
   150→    // Check win after timeout elimination
   151→    const winCheck = game.checkWin();
   152→    if (winCheck) {
   153→      const winEmbed = new EmbedBuilder()
   154→        .setColor(winCheck.winner === 'wolves' ? 0xE74C3C : 0x2ECC71)
   155→        .setTitle(winCheck.winner === 'wolves' ? 'Wolves Win!' : 'Village Wins!')
   156→        .setDescription(`${winCheck.message}\n\n${game.getFullPlayerListString()}`)
   157→        .setFooter({ text: 'Game Over — Thanks for playing!' })
   158→        .setTimestamp();
   159→      await channel.send({ embeds: [winEmbed] });
   160→      activeGames.delete(channel.id);
   161→      return;
   162→    }
   163→
   164→    // Pass gun to a random alive player
   165→    const alivePlayers = game.getAlivePlayers();
   166→    if (alivePlayers.length > 0) {
   167→      const nextHolder = alivePlayers[Math.floor(Math.random() * alivePlayers.length)];
   168→      game.gunHolder = nextHolder.user.id;
   169→
   170→      const passEmbed = new EmbedBuilder()
   171→        .setColor(0xFFD700)
   172→        .setTitle('🔫 Gun Passed!')
   173→        .setDescription(
   174→          `The gun passes to **${nextHolder.user.username}**!\n\n` +
   175→          `Use \`w.shoot <number>\` to shoot!\n⏱️ **${game.shootTimerLength}s** on the clock!`
   176→        )
   177→        .setTimestamp();
   178→      await channel.send({ embeds: [passEmbed] });
   179→
   180→      // Start new timer for the next gun holder
   181→      await startPopcornShootTimer(game);
   182→    }
   183→  }, game.shootTimerLength * 1000);
   184→}
